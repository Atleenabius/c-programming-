C Program: Addition of Two Numbers

#include <stdio.h>

int main() {
    int a = 10, b = 20, sum;

    sum = a + b;

    printf("First number = %d\n", a);
    printf("Second number = %d\n", b);
    printf("Sum = %d\n", sum);

    return 0;
}
